defineFunction(CountExperts, int)
{
	fn = 0;
	Script_SetUpdateFrequency(20000);
	Person_GetNumberOfPeople(cn);
	for (ci=0, ci<cn, ci=ci+1)
	{
		wor.GetDataByIndex(ci);
		if (wor.nValidRead & wor.fEducation > 2.999)
		{
			fn = fn + 1;
		}
	}
	Script_SetUpdateFrequency(5000);
	return(fn);
}

defineFunction(CountTestStands, int)
{
	fk = 0;
	Building_GetNumberOfBuildings(bn);
	for (bi=0, bi<bn, bi=bi+1)
	{
		bui.GetDataByIndex(bi);
		if (bui.nValidRead & bui.nType ? BUILDINGTYPE_FACTORY & bui.nWorkersNeeded ? 80 & bui.nProffesorsNeeded ? 40 & bui.fPercFinished > 0.9999 & bui.nWorkersNum > 0)
		{
			fk = fk + 1;
		}
	}
	return(fk);
}

defineFunction(FindRocket, int, float:fPowerWanted)
{
	fi = 0 - 1;
	Vehicle_GetNumberOfVehicles(tn);
	for (ti=0, ti<tn, ti=ti+1)
	{
		vehi.GetDataByIndex(ti);
		if (vehi.nValidRead & vehi.nVehicleType ? VEHICLETYPE_HELICOPTER & vehi.fType_EnginePower > fPowerWanted - 0.5 & vehi.fType_EnginePower < fPowerWanted + 0.5)
		{
			fj = vehi.nBuilding_HomeWorkplaceID;
			if (fj > -1)
			{
				bui.GetDataByIndex(fj);
				if (bui.nValidRead & bui.nType ? BUILDINGTYPE_AIRPLANE_PARKING)
				{
					fDist = DistancePoints2D(vehi.vPosition, bui.vPosition);
					nPadOK = 1;
					if (fj ? nBlockedPad)
					{
						nNow = year * 365 + day;
						if (nNow < nBlockUntil)
						{
							nPadOK = 0;
						}
					}
					if (fDist < 90.00 & nPadOK)
					{
						fi = ti;
						nPad = fj;
						padPos = bui.vPosition;
					}
				}
			}
		}
	}
	return(fi);
}

defineFunction(SumNear, void)
{
	fFuel = 0;
	fLox = 0;
	fHyper = 0;
	fCraft = 0;
	fFood = 0;
	Script_SetUpdateFrequency(20000);
	Building_GetNumberOfBuildings(sn);
	for (si=0, si<sn, si=si+1)
	{
		bui.GetDataByIndex(si);
		if (bui.nValidRead & bui.nType ? BUILDINGTYPE_STORAGE & bui.nStorageNum > 0)
		{
			fDist = DistancePoints2D(bui.vPosition, padPos);
			if (fDist < 450.00)
			{
				res.ResetAmounts();
				res.GetFromBuilding(si);
				fFuel = fFuel + res.fuel;
				fLox = fLox + res._Resources_reserved_16_;
				fHyper = fHyper + res._Resources_reserved_17_;
				fCraft = fCraft + res._Resources_reserved_18_;
				fFood = fFood + res.food;
			}
		}
	}
	Script_SetUpdateFrequency(5000);
	returnVoid();
}

defineFunction(IsResearched, int, string:sResearchKey)
{
	gs.GetCurrentGameSettigns();
	if (!gs.Research)
	{
		return(1);
	}
	rr = 0;
	Research_IsCompleted(sResearchKey, rr);
	return(rr);
}

defineFunction(FindLaunched, int, float:fPowerLaunched)
{
	li = 0 - 1;
	lk = 0;
	Vehicle_GetNumberOfVehicles(ln);
	for (ki=0, ki<ln, ki=ki+1)
	{
		vehi.GetDataByIndex(ki);
		if (vehi.nValidRead & vehi.nVehicleType ? VEHICLETYPE_HELICOPTER & vehi.fType_EnginePower > fPowerLaunched - 0.5 & vehi.fType_EnginePower < fPowerLaunched + 0.5)
		{
			vtmp = vehi.vPosition;
			fUp = vtmp.y - padPos.y;
			if (vehi.bDisableSell)
			{
				li = ki;
				lk = 1;
			}
			else()
			{
				if (fUp > 20.0)
				{
					li = ki;
				}
			}
		}
	}
	return(li);
}

defineFunction(CheckCosmonauts, void)
{
	if (bObjReady)
	{
		nCosmo = 0;
		Script_SetUpdateFrequency(20000);
		Person_GetNumberOfPeople(kn);
		for (ki=0, ki<kn, ki=ki+1)
		{
			wor.GetDataByIndex(ki);
			if (wor.nValidRead & wor.fEducation > 2.999)
			{
				nCosmo = nCosmo + 1;
				cosmoPos = wor.vPosition;
			}
		}
		Script_SetUpdateFrequency(5000);
		if (!bCosmoObj)
		{
			Objectives_CreateNewString("sr_cosmo", "Experts: cosmonauts and chief designers");
			Objective_AddRequirement("sr_cosmo", 3.0, "research/sr_manned_flight.png");
			bCosmoObj = 1;
		}
		fCosmo = nCosmo;
		Objective_UpdateRequirement("sr_cosmo", 0, fCosmo);
		if (nCosmo > nCosmoShown)
		{
			Notification_CreateNewStringPic("A new expert", "Star City has trained a cosmonaut, or OKB-1 a chief designer. The objective counts every expert in the republic.", "research/sr_manned_flight.png", cosmoPos);
			nCosmoShown = nCosmo;
		}
	}
	returnVoid();
}

defineFunction(BoostLoyalty, void, float:fDelta)
{
	Script_SetUpdateFrequency(20000);
	Person_GetNumberOfPeople(cn);
	for (ci=0, ci<cn, ci=ci+1)
	{
		wor.GetDataByIndex(ci);
		if (wor.nValidRead)
		{
			f2 = wor.fStatusSoviet + fDelta;
			if (f2 > 1.0)
			{
				f2 = 1.0;
			}
			if (f2 < 0.0)
			{
				f2 = 0.0;
			}
			Person_SetStatus(ci, 3, f2);
		}
	}
	Script_SetUpdateFrequency(5000);
	returnVoid();
}

defineFunction(main, void)
{
	InitConstants();
	bObjReady = 0;
	bCosmoObj = 0;
	nCosmoShown = 0;
	nBlockUntil = 0;
	nBlockedPad = 0 - 1;
	for (i=0, i<8, i=i+1)
	{
		srDone[i] = 0;
		usDone[i] = 0;
	}
	for (i=0, i<6, i=i+1)
	{
		rocketOK[i] = 0;
	}
	// wait quietly for the first space research - an ordinary republic never notices this script
	nResearch = 0;
	while (!nResearch)
	{
		Script_Sleep(30.0);
		nResearch = IsResearched("sr_rocketry");
		CheckUSA();
	}
	Scenario_WindowWithImageLeft("The Space Programme", "The Rocket Research Institute has opened. Korolev's designers promise a satellite before the Americans, then a man in space, then the Moon. Research the space branch, build the plants, the launch complexes and the cosmonaut corps. The Americans will not wait: they mean to land on the Moon by 20 July 1969.", "programme.png", 3);
	winexist = 1;
	while (winexist)
	{
		Scenario_WindowExists(winexist);
	}
	Scenario_ObjectiveCreate("sr_race", "The Space Race", "Reach every milestone before the United States. The last one: an N1 carrying cosmonauts to the Moon before the Americans get there, by 20 July 1969.");
	bObjReady = 1;

	// ---------------------------------------------------------------- sputnik
	Objectives_CreateNewString("sr_next", "Next: The first satellite - research Artificial Earth Satellite");
	Objective_AddRequirement("sr_next", 1.0, "research/sr_satellite.png");
	nResearch = 0;
	while (!nResearch)
	{
		Script_Sleep(20.0);
		nResearch = IsResearched("sr_satellite");
		CheckUSA();
	}
	Objective_Remove("sr_next");
	Scenario_WindowWithImageLeft("The first satellite", "Put an artificial satellite into orbit. Park an R-7 Sputnik rocket on a launch complex, store kerosene, liquid oxygen and the satellite in storage buildings within 450 m of the pad, and it will fly.", "sputnik.png", 3);
	winexist = 1;
	while (winexist)
	{
		Scenario_WindowExists(winexist);
	}
	nNeedCrew = 0;
	nNeedTrack = 0;
	Objectives_CreateNewString("sr_rocket", "R-7 Sputnik on a launch pad");
	Objective_AddRequirement("sr_rocket", 1.0, "research/sr_satellite.png");
	Objectives_CreateNewString("sr_prop", "Propellant near the pad (t)");
	Objective_AddRequirement("sr_prop", 28.00, "resources/fuel.png");
	Objectives_CreateNewString("sr_craft", "Payload near the pad (t)");
	Objective_AddRequirement("sr_craft", 2.00, "resources/spacecraft.png");
	if (nNeedCrew > 0)
	{
		Objectives_CreateNewString("sr_crew", "Cosmonauts (experts)");
		f = nNeedCrew;
		Objective_AddRequirement("sr_crew", f, "research/sr_manned_flight.png");
	}
	if (nNeedTrack > 0)
	{
		Objectives_CreateNewString("sr_track", "Tracking stations");
		f = nNeedTrack;
		Objective_AddRequirement("sr_track", f, "research/sr_satellite.png");
	}
	while (!srDone[0])
	{
		Script_Sleep(10.0);
		CheckUSA();
		bContinue = 1;
		nVeh = FindRocket(9001.0);
		if (nVeh > -1)
		{
			Objective_UpdateRequirement("sr_rocket", 0, 1.0);
		}
		else()
		{
			Objective_UpdateRequirement("sr_rocket", 0, 0.0);
			bContinue = 0;
		}
		if (bContinue)
		{
			SumNear();
			f = 0;
			fNeed = 8.00 - 0.01;
			if (fFuel > fNeed)
			{
				f = f + 8.00;
			}
			else()
			{
				f = f + fFuel;
				bContinue = 0;
			}
			fNeed = 20.00 - 0.01;
			if (fLox > fNeed)
			{
				f = f + 20.00;
			}
			else()
			{
				f = f + fLox;
				bContinue = 0;
			}
			fNeed = 0.00 - 0.01;
			if (fHyper > fNeed)
			{
				f = f + 0.00;
			}
			else()
			{
				f = f + fHyper;
				bContinue = 0;
			}
			Objective_UpdateRequirement("sr_prop", 0, f);
			f = fCraft;
			if (f > 2.00)
			{
				f = 2.00;
			}
			Objective_UpdateRequirement("sr_craft", 0, f);
			fNeed = 2.00 - 0.01;
			if (fCraft < fNeed)
			{
				bContinue = 0;
			}
			fNeed = 1.00 - 0.01;
			if (fFood < fNeed)
			{
				bContinue = 0;
			}
		}
		if (nNeedCrew > 0)
		{
			nExperts = CountExperts();
			f = nExperts;
			Objective_UpdateRequirement("sr_crew", 0, f);
			if (nExperts < nNeedCrew)
			{
				bContinue = 0;
			}
		}
		if (nNeedTrack > 0)
		{
			nTrack = CountTracking();
			f = nTrack;
			Objective_UpdateRequirement("sr_track", 0, f);
			if (nTrack < nNeedTrack)
			{
				bContinue = 0;
			}
		}
		if (bContinue)
		{
			// launch: the spacerace plugin takes the load from the storages and flies the flagged rocket
			Scenario_ObjectiveMoveCameraTo(padPos, 260.0);
			Notification_CreateNewStringPic("Launch", "R-7 Sputnik ignition... lift-off!", "research/sr_satellite.png", padPos);
			Script_Sleep(4.0);
			fFail = 30 - 5 * rocketOK[0];
			nTest = CountTestStands();
			if (nTest > 0)
			{
				fFail = fFail - 10;
			}
			if (fFail < 5)
			{
				fFail = 5;
			}
			Random(r);
			r = r % 100;
			fr = r;
			Vehicle_SetCanSell(nVeh, 0);
			if (fr < fFail)
			{
				Script_Sleep(3.0);
				nVeh = FindLaunched(9001.0);
				if (nVeh > -1)
				{
					Vehicle_Sell(nVeh, 0, 0);
				}
				gs.GetCurrentGameSettigns();
				if (gs.FiresEnabled)
				{
					Building_StartFire(nPad);
					Notification_CreateNewStringPic("Launch failure", "The R-7 Sputnik exploded. The pad is burning: send the fire brigade, then build another rocket and try again.", "research/sr_satellite.png", padPos);
				}
				else()
				{
					Date_GetCurrentDate_D365Y(day, year);
					nBlockedPad = nPad;
					nBlockUntil = year * 365 + day + 30;
					Notification_CreateNewStringPic("Launch failure", "The R-7 Sputnik exploded and wrecked the pad. Repairs take 30 days; build another rocket meanwhile.", "research/sr_satellite.png", padPos);
				}
			}
			else()
			{
				// the plugin clears the flag when the climb is over (at most a minute)
				k = 0;
				bContinue = 1;
				while (bContinue)
				{
					Script_Sleep(1.0);
					k = k + 1;
					nVeh = FindLaunched(9001.0);
					if (nVeh < 0 | !lk | k > 60)
					{
						bContinue = 0;
					}
				}
				if (nVeh > -1)
				{
					Vehicle_Sell(nVeh, 0, 0);
				}
				srDone[0] = 1;
				rocketOK[0] = rocketOK[0] + 1;
				if (usDone[0])
				{
					Money_AddUSD(5000);
					fLoyal = 0.03;
					BoostLoyalty(fLoyal);
					Scenario_WindowWithImageLeft("The first satellite", "Beep... beep... beep. Sputnik is in orbit and every radio on Earth can hear it. The space age has begun, and it began here. The Americans did it first, but we are catching up.", "sputnik.png", 3);
				}
				else()
				{
					Money_AddUSD(25000);
					fLoyal = 0.12;
					BoostLoyalty(fLoyal);
					Scenario_WindowWithImageLeft("The first satellite", "Beep... beep... beep. Sputnik is in orbit and every radio on Earth can hear it. The space age has begun, and it began here. A Soviet first: the world is watching.", "sputnik.png", 3);
				}
				winexist = 1;
				while (winexist)
				{
					Scenario_WindowExists(winexist);
				}
			}
		}
	}
	Objective_Remove("sr_rocket");
	Objective_Remove("sr_prop");
	Objective_Remove("sr_craft");
	if (nNeedCrew > 0)
	{
		Objective_Remove("sr_crew");
	}
	if (nNeedTrack > 0)
	{
		Objective_Remove("sr_track");
	}

	// ---------------------------------------------------------------- laika
	Objectives_CreateNewString("sr_next", "Next: A passenger in orbit - research Biological Satellites");
	Objective_AddRequirement("sr_next", 1.0, "research/sr_biosatellite.png");
	nResearch = 0;
	while (!nResearch)
	{
		Script_Sleep(20.0);
		nResearch = IsResearched("sr_biosatellite");
		CheckUSA();
	}
	Objective_Remove("sr_next");
	Scenario_WindowWithImageLeft("A passenger in orbit", "Send a living passenger into orbit: a satellite with life support and food for the flight. Belka and Strelka came back; Laika did not.", "laika.png", 3);
	winexist = 1;
	while (winexist)
	{
		Scenario_WindowExists(winexist);
	}
	nNeedCrew = 0;
	nNeedTrack = 0;
	Objectives_CreateNewString("sr_rocket", "R-7 Sputnik on a launch pad");
	Objective_AddRequirement("sr_rocket", 1.0, "research/sr_biosatellite.png");
	Objectives_CreateNewString("sr_prop", "Propellant near the pad (t)");
	Objective_AddRequirement("sr_prop", 28.00, "resources/fuel.png");
	Objectives_CreateNewString("sr_craft", "Payload near the pad (t)");
	Objective_AddRequirement("sr_craft", 2.00, "resources/spacecraft.png");

			Notification_CreateNewStringPic("Launch", "N1-L3 ignition... lift-off!", "research/sr_lunar_landing.png", padPos);
			Script_Sleep(4.0);
			fFail = 70 - 5 * rocketOK[4];
			r = IsResearched("sr_nk33");
			if (r)
			{
				fFail = fFail - 35;
			}
			nTest = CountTestStands();
			if (nTest > 0)
			{
				fFail = fFail - 10;
			}
			if (fFail < 5)
			{
				fFail = 5;
			}
			Random(r);
			r = r % 100;
			fr = r;
			Vehicle_SetCanSell(nVeh, 0);
			if (fr < fFail)
			{
				Script_Sleep(3.0);
				nVeh = FindLaunched(9005.0);
				if (nVeh > -1)
				{
					Vehicle_Sell(nVeh, 0, 0);
				}
				gs.GetCurrentGameSettigns();
				if (gs.FiresEnabled)
				{
					Building_StartFire(nPad);
					Notification_CreateNewStringPic("Launch failure", "The N1-L3 exploded. The pad is burning: send the fire brigade, then build another rocket and try again.", "research/sr_lunar_landing.png", padPos);
				}
				else()
				{
					Date_GetCurrentDate_D365Y(day, year);
					nBlockedPad = nPad;
					nBlockUntil = year * 365 + day + 30;
					Notification_CreateNewStringPic("Launch failure", "The N1-L3 exploded and wrecked the pad. Repairs take 30 days; build another rocket meanwhile.", "research/sr_lunar_landing.png", padPos);
				}
			}
			else()
			{
				// the plugin clears the flag when the climb is over (at most a minute)
				k = 0;
				bContinue = 1;
				while (bContinue)
				{
					Script_Sleep(1.0);
					k = k + 1;
					nVeh = FindLaunched(9005.0);
					if (nVeh < 0 | !lk | k > 60)
					{
						bContinue = 0;
					}
				}
				if (nVeh > -1)
				{
					Vehicle_Sell(nVeh, 0, 0);
				}
				srDone[7] = 1;
				rocketOK[4] = rocketOK[4] + 1;
				if (usDone[7])
				{
					Money_AddUSD(5000);
					fLoyal = 0.03;
					BoostLoyalty(fLoyal);
					Scenario_WindowWithImageLeft("A Soviet footprint on the Moon", "The LK has landed. A Soviet cosmonaut stands on the Moon. The Americans did it first, but we are catching up.", "moon.png", 3);
				}
				else()
				{
					Money_AddUSD(25000);
					fLoyal = 0.12;
					BoostLoyalty(fLoyal);
					Scenario_WindowWithImageLeft("A Soviet footprint on the Moon", "The LK has landed. A Soviet cosmonaut stands on the Moon. A Soviet first: the world is watching.", "moon.png", 3);
				}
				winexist = 1;
				while (winexist)
				{
					Scenario_WindowExists(winexist);
				}
			}
		}
	}
	Objective_Remove("sr_rocket");
	Objective_Remove("sr_prop");
	Objective_Remove("sr_craft");
	if (nNeedCrew > 0)
	{
		Objective_Remove("sr_crew");
	}
	if (nNeedTrack > 0)
	{
		Objective_Remove("sr_track");
	}

	if (usDone[7])
	{
		Scenario_WindowWithImageLeft("The race is over", "We reached the Moon, but the Americans were there first. History will remember Apollo 11 - and the N1 that came after it.", "moon.png", 3);
	}
	else()
	{
		Scenario_WindowWithImageLeft("Victory in the Space Race", "A Soviet cosmonaut walked on the Moon before any American. From Sputnik to the N1, the republic won the race to space.", "moon.png", 3);
		Money_AddUSD(100000);
		fLoyal = 0.20;
		BoostLoyalty(fLoyal);
	}
	winexist = 1;
	while (winexist)
	{
		Scenario_WindowExists(winexist);
	}
	Scenario_ObjectiveSetCompleted("sr_race", 0, 1);
	Scenario_UnlockNextScenarios();
}
