defineFunction(CountExperts, int)
{
	fn = 0;
	Script_SetUpdateFrequency(20000);
	Person_GetNumberOfPeople(cn);
	for (ci=0, ci<cn, ci=ci+1)
	{
		wor.GetDataByIndex(ci);
		if (wor.nValidRead & wor.fEducation > 2.999)
		{
			fn = fn + 1;
		}
	}
	Script_SetUpdateFrequency(5000);
	return(fn);
}

defineFunction(CountTracking, int)
{
	fk = 0;
	Building_GetNumberOfBuildings(bn);
	for (bi=0, bi<bn, bi=bi+1)
	{
		bui.GetDataByIndex(bi);
		if (bui.nValidRead & bui.nType ? BUILDINGTYPE_BROADCAST & bui.nProffesorsNeeded ? 30 & bui.nWorkersNeeded ? 40 & bui.fPercFinished > 0.9999)
		{
			fk = fk + 1;
		}
	}
	return(fk);
}

defineFunction(FindRocket, int, float:fPowerWanted)
{
	fi = 0 - 1;
	Vehicle_GetNumberOfVehicles(tn);
	for (ti=0, ti<tn, ti=ti+1)
	{
		vehi.GetDataByIndex(ti);
		if (vehi.nValidRead & vehi.nVehicleType ? VEHICLETYPE_HELICOPTER & vehi.fType_EnginePower > fPowerWanted - 0.5 & vehi.fType_EnginePower < fPowerWanted + 0.5)
		{
			fj = vehi.nBuilding_HomeWorkplaceID;
			if (fj > -1)
			{
				bui.GetDataByIndex(fj);
				if (bui.nValidRead & bui.nType ? BUILDINGTYPE_AIRPLANE_PARKING)
				{
					fDist = DistancePoints2D(vehi.vPosition, bui.vPosition);
					if (fDist < 90.0)
					{
						fi = ti;
						nPad = fj;
						padPos = bui.vPosition;
					}
				}
			}
		}
	}
	return(fi);
}

defineFunction(SumNear, void)
{
	fFuel = 0;
	fLox = 0;
	fCraft = 0;
	fFood = 0;
	Script_SetUpdateFrequency(20000);
	Building_GetNumberOfBuildings(sn);
	for (si=0, si<sn, si=si+1)
	{
		bui.GetDataByIndex(si);
		if (bui.nValidRead & bui.nStorageNum > 0)
		{
			fDist = DistancePoints2D(bui.vPosition, padPos);
			if (fDist < 450.0)
			{
				res.ResetAmounts();
				res.GetFromBuilding(si);
				fFuel = fFuel + res.fuel;
				fLox = fLox + res.chemicals;
				fCraft = fCraft + res.eletronics;
				fFood = fFood + res.food;
			}
		}
	}
	Script_SetUpdateFrequency(5000);
	returnVoid();
}

defineFunction(TakeNear, void)
{
	Building_GetNumberOfBuildings(un);
	for (ui=0, ui<un, ui=ui+1)
	{
		bui.GetDataByIndex(ui);
		if (bui.nValidRead & bui.nStorageNum > 0)
		{
			fDist = DistancePoints2D(bui.vPosition, padPos);
			if (fDist < 450.0)
			{
				res.ResetAmounts();
				res.GetFromBuilding(ui);
				rtake.ResetAmounts();
				fTake = res.fuel;
				if (fTake > fNeedFuel)
				{
					fTake = fNeedFuel;
				}
				if (fTake > 0)
				{
					rtake.fuel = 0 - fTake;
					fNeedFuel = fNeedFuel - fTake;
				}
				fTake = res.chemicals;
				if (fTake > fNeedLox)
				{
					fTake = fNeedLox;
				}
				if (fTake > 0)
				{
					rtake.chemicals = 0 - fTake;
					fNeedLox = fNeedLox - fTake;
				}
				fTake = res.eletronics;
				if (fTake > fNeedCraft)
				{
					fTake = fNeedCraft;
				}
				if (fTake > 0)
				{
					rtake.eletronics = 0 - fTake;
					fNeedCraft = fNeedCraft - fTake;
				}
				fTake = res.food;
				if (fTake > fNeedFood)
				{
					fTake = fNeedFood;
				}
				if (fTake > 0)
				{
					rtake.food = 0 - fTake;
					fNeedFood = fNeedFood - fTake;
				}
				rtake.AddFromBuilding(ui);
			}
		}
	}
	returnVoid();
}

defineFunction(BoostLoyalty, void, float:fDelta)
{
	Script_SetUpdateFrequency(20000);
	Person_GetNumberOfPeople(cn);
	for (ci=0, ci<cn, ci=ci+1)
	{
		wor.GetDataByIndex(ci);
		if (wor.nValidRead)
		{
			f2 = wor.fStatusSoviet + fDelta;
			if (f2 > 1.0)
			{
				f2 = 1.0;
			}
			if (f2 < 0.0)
			{
				f2 = 0.0;
			}
			Person_SetStatus(ci, 3, f2);
		}
	}
	Script_SetUpdateFrequency(5000);
	returnVoid();
}

	Objective_Remove("sr_prop");
	Objective_Remove("sr_craft");
	Objective_Remove("sr_track");

	// ---------------------------------------------------------------- moon
	nResearch = 0;
	while (!nResearch)
	{
		Script_Sleep(20.0);
		Research_IsCompleted("sr_lunar_landing", nResearch);
		CheckUSA();
	}
	Scenario_WindowWithImageLeft("A Soviet footprint on the Moon", "The N1 has to fly. Park it on the Heavy Launch Complex with 80 t of kerosene, 200 t of liquid oxygen and 25 t of spacecraft in reach, two cosmonauts and three tracking stations. Without the NK-33 engines most N1s explode.", "moon.png", 3);
	winexist = 1;
	while (winexist)
	{
		Scenario_WindowExists(winexist);
	}
	Objectives_CreateNewString("sr_rocket", "N1-L3 on a launch pad");
	Objective_AddRequirement("sr_rocket", 1.0, "research/sr_lunar_landing.png");
	Objectives_CreateNewString("sr_prop", "Propellant near the pad (t)");
	Objective_AddRequirement("sr_prop", 280.0, "resources/fuel.png");
	Objectives_CreateNewString("sr_craft", "Payload near the pad (t)");
	Objective_AddRequirement("sr_craft", 25.0, "resources/eletronics.png");
	Objectives_CreateNewString("sr_crew", "Cosmonauts (experts)");
	Objective_AddRequirement("sr_crew", 2.0, "research/sr_manned_flight.png");
	Objectives_CreateNewString("sr_track", "Tracking stations");
	Objective_AddRequirement("sr_track", 3.0, "research/sr_satellite.png");
	while (!srDone[7])
	{
		Script_Sleep(10.0);
		CheckUSA();
		bContinue = 1;
		nVeh = FindRocket(9005.0);
		if (nVeh > -1)
		{
			Objective_UpdateRequirement("sr_rocket", 0, 1.0);
		}
		else()
		{
			Objective_UpdateRequirement("sr_rocket", 0, 0.0);
			bContinue = 0;
		}
		if (bContinue)
		{
			SumNear();
			f = 0;
			if (fFuel > 80.0)
			{
				f = f + 80.0;
			}
			else()
			{
				f = f + fFuel;
				bContinue = 0;
			}
			if (fLox > 200.0)
			{
				f = f + 200.0;
			}
			else()
			{
				f = f + fLox;
				bContinue = 0;
			}
			Objective_UpdateRequirement("sr_prop", 0, f);
			f = fCraft;
			if (f > 25.0)
			{
				f = 25.0;
			}
			Objective_UpdateRequirement("sr_craft", 0, f);
			if (fCraft < 25.0)
			{
				bContinue = 0;
			}
			if (fFood < 3.0)
			{
				bContinue = 0;
			}
		}
		nExperts = CountExperts();
		f = nExperts;
		Objective_UpdateRequirement("sr_crew", 0, f);
		if (nExperts < 2)
		{
			bContinue = 0;
		}
		nTrack = CountTracking();
		f = nTrack;
		Objective_UpdateRequirement("sr_track", 0, f);
		if (nTrack < 3)
		{
			bContinue = 0;
		}
		if (bContinue)
		{
			// launch: the goods are spent whatever happens
			fNeedFuel = 80.0;
			fNeedLox = 200.0;
			fNeedCraft = 25.0;
			fNeedFood = 3.0;
			TakeNear();
			Scenario_ObjectiveMoveCameraTo(padPos, 260.0);
			Notification_CreateNewStringPic("Launch", "N1-L3 ignition... lift-off!", "research/sr_lunar_landing.png", padPos);
			Script_Sleep(4.0);
			fFail = 70 - 5 * rocketOK[4];
			Research_IsCompleted("sr_nk33", r);
			if (r)
			{
				fFail = fFail - 35;
			}
			if (fFail < 5)
			{
				fFail = 5;
			}
			Random(r);
			r = r % 100;
			fr = r;
			Vehicle_Sell(nVeh, 0, 0);
			if (fr < fFail)
			{
				Building_StartFire(nPad);
				prestige = prestige - 2;
				Notification_CreateNewStringPic("Launch failure", "The N1-L3 exploded. The pad is burning. Build another rocket and try again.", "research/sr_lunar_landing.png", padPos);
			}
			else()
			{
				srDone[7] = 1;
				rocketOK[4] = rocketOK[4] + 1;
				if (usDone[7])
				{
					prestige = prestige + 3;
					Money_AddUSD(5000);
					fLoyal = 0.03;
					BoostLoyalty(fLoyal);
					Scenario_WindowWithImageLeft("A Soviet footprint on the Moon", "The LK has landed. A Soviet cosmonaut stands on the Moon. The Americans did it first, but we are catching up.", "moon.png", 3);
				}
				else()
				{
					prestige = prestige + 10;
					nFirsts = nFirsts + 1;
					Money_AddUSD(25000);
					fLoyal = 0.12;
					BoostLoyalty(fLoyal);
					Scenario_WindowWithImageLeft("A Soviet footprint on the Moon", "The LK has landed. A Soviet cosmonaut stands on the Moon. A Soviet first: the world is watching.", "moon.png", 3);
				}
				winexist = 1;
				while (winexist)
				{
					Scenario_WindowExists(winexist);
				}
			}
		}
	}
	Objective_Remove("sr_rocket");
	Objective_Remove("sr_prop");
	Objective_Remove("sr_craft");
	Objective_Remove("sr_crew");
	Objective_Remove("sr_track");

	if (usDone[7])
	{
		Scenario_WindowWithImageLeft("The race is over", "We reached the Moon, but the Americans were there first. History will remember Apollo 11 - and the N1 that came after it.", "moon.png", 3);
	}
	else()
	{
		Scenario_WindowWithImageLeft("Victory in the Space Race", "A Soviet cosmonaut walked on the Moon before any American. From Sputnik to the N1, the republic won the race to space.", "moon.png", 3);
		Money_AddUSD(100000);
		fLoyal = 0.2;
		BoostLoyalty(fLoyal);
	}
	winexist = 1;
	while (winexist)
	{
		Scenario_WindowExists(winexist);
	}
	Scenario_ObjectiveSetCompleted("sr_race", 0, 1);
	Scenario_UnlockNextScenarios();
}
